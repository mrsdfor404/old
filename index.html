// Select NO button and white card container
const noBtn = document.getElementById('no-btn');
const card = document.getElementById('card');

// Bounded escape physics inside the card
function dodgeInsideCard() {
    const cardRect = card.getBoundingClientRect();
    const padding = 20;
    
    const maxX = cardRect.width - noBtn.offsetWidth - padding;
    const maxY = cardRect.height - noBtn.offsetHeight - padding;

    const newX = Math.random() * (maxX - padding) + padding;
    const newY = Math.random() * (maxY - padding) + padding;

    noBtn.style.position = 'absolute';
    noBtn.style.transform = `translate3d(${newX}px, ${newY}px, 0)`;
}
